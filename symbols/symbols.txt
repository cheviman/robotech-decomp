# Robotech: Battlecry (Xbox, XDK 4721) - the symbols of default.xbe
#
# Addresses and names only: no game data or code lives in this file. tools/symbols.py reads and writes
# it; tools/apply_names.py puts the names into the disassembly, the recompiler emits them, and
# tools/organize_gen.py lays the C out in files with each function's description.
#
# Format:  ADDRESS  KIND  NAME  [# comment]
#   ADDRESS  the Xbox virtual address, 8 hexadecimal digits
#   KIND     func = a function's entry, data = a variable or table
#   ADDRESS  file  PATH  [# what the stretch holds]
#            the functions from ADDRESS up to the next file line go to src/recomp/gen/PATH.c; a file may
#            take several stretches, and the comment of its first line describes it
#   comment  NAME  text
#            what the function NAME does, in the game's terms and with the C's names
#
# A name is real -- it says what the thing is -- or there is none: a type's number, a slot's offset or an
# address is not a name (tools/noise_report.py counts any that reaches the C).
#
# Sections headed "(generated by ...)" are rewritten by tools/symbols.py; every other line is written by
# hand, and a hand line wins over a generated one for the same address.

# The object system's structures, as its functions read them; a field nobody has seen used is not listed.
struct  vcClassDesc  5C                         # what a class registers: vcRegisterClass copies it into its vcClass
field   vcClassDesc  00  type      u32          # the class's type: its index in vcClassTable and vcClassByType, a handle's high 16 bits
field   vcClassDesc  04  name      char[32]     # the class's name: vcTexture; the folder of its files
field   vcClassDesc  24  file_tag  char[4]      # the extension of its files: txr
field   vcClassDesc  28  version   u32          # the newest version of the object its load callback reads
field   vcClassDesc  2C  chunk_version  u32     # the newest version of a chunk its load_chunk callback reads
field   vcClassDesc  30  flags     u32          # 1: its objects are files of their own, with names unique in the class, which vcObjectCollectChanged lists to be saved apart; 4: vcObjectGetShared clones instead of sharing; 8: no path is reported for a missing object
field   vcClassDesc  34  create    u32          # create(handle): allocates the object, which it hands back
field   vcClassDesc  38  destroy   u32          # destroy(handle, object): frees the object and lets go of its children
field   vcClassDesc  3C  collect   u32          # collect(handle, object, changed, list): collects the children, hands back the changed bits with theirs
field   vcClassDesc  40  load      u32          # load(handle, file, version, object): reads the object, by the version
field   vcClassDesc  44  reload    u32          # reload(handle, file, version, object): reads an object that already exists again
field   vcClassDesc  48  load_chunk  u32        # load_chunk(handle, file, version, object): reads one chunk of the object
field   vcClassDesc  4C  copy      u32          # copy(handle, object): fills a new object from another
field   vcClassDesc  50  in_use    u32          # in_use(handle, object, time): nonzero while the object is still in use at that time
field   vcClassDesc  54  get_size  u32          # get_size(handle, object, what): the object's size in bytes
struct  vcClass  70                             # a class in vcClassTable: its vcClassDesc (whose fields sit at the same offsets), then its instances
field   vcClass  5C  free_index  s32            # where the next instance is taken from
field   vcClass  60  count       s32            # the instances in use
field   vcClass  64  capacity    s32            # the instances the table has room for: vcRegisterClass's second argument, doubled when it fills
field   vcClass  68  peak        s32            # the most instances in use at once
field   vcClass  6C  instances   u32  vcInstance # its table of instances, capacity x 0x4C bytes
struct  vcInstance  4C                          # an object of a class: a handle's low 16 bits index it in the class's instances
field   vcInstance  00  object  u32             # the object's own data, which the class's callbacks work on; 0 for a free instance
field   vcInstance  04  refs    s32             # the references to it; at 0 it is destroyed
field   vcInstance  08  flags   u32             # bit 0: its data changed; bit 1: its name; bit 2: it is new (the three set at creation, cleared once it is read from a file); bit 4: in vcReleaseQueue
field   vcInstance  0C  name    char[64]        # its name, lower case, unique among its class's if the class says so
struct  vcCollectList  8                        # what vcObjectCollectChanged fills: the handles of the objects to save
field   vcCollectList  00  whole   u32          # nonzero: every object goes in, changed or not
field   vcCollectList  04  count   u32          # the handles in it, which follow from offset 8


comment vcObjectSystemInit Starts the object system: no class registered, the folder the objects' files are read from, and the names of the platforms whose folders they may be in.
comment vcClassFindFreeInstance Sets the class's free_index to a free instance: the first from free_index on, else the last before it, else the end of the table.
comment vcGetDataDir The folder the objects' files are under (vcDataDir).
comment vcGetPlatformName The folder of this platform's files under each class's: "xbox".
comment vcRegisterClass Registers an engine class: copies its descriptor (the type, its name, its file tag and the callbacks the object system calls) into vcClassTable, points vcClassByType at it and allocates its table of instances.
comment vcObjectCreate Creates an object of a class: takes a free instance (the table doubles when it is full), marks it changed with one reference and runs the class's create callback. Hands back the new handle.
comment vcObjectAddRef Adds a reference to an object, if the handle is not 0. Hands the handle back.
comment vcClassAddRefAll Adds a reference to every object of a class that has one.
comment vcObjectRelease Takes a reference from an object; at none left, the class's destroy callback frees it and the instance is free again.
comment vcClassDecRefAll Takes a reference from every object of a class that has one, destroying none.
comment vcObjectGetName The object's name.
comment vcObjectMarkChanged Marks the object's data changed.
comment vcObjectClearChanged Marks the object as read from its file: nothing of it changed, nor new.
comment vcObjectFind The object of a class with that name (no name: the first one in use), from the given handle's index on, or among those in use. 0 if there is none.
comment vcClassForEach Calls a function over the objects of a class in use, at most the given number of them (any, if negative), until it hands back 0. Hands back 0 if it stopped so, 1 otherwise.
comment vcObjectClone Makes a new object of the same class as the given one and fills it from it with the class's copy callback.
comment vcObjectGetSize The object's size in bytes, from its class's get_size callback (0 if it has none).
comment vcObjectCollectChanged Collects what has to be saved: an object of a class whose objects are files of their own goes in the list when it or one of its children changed (its class's collect callback goes over the children); the others are saved with whoever refers to them. Hands back whether that one changed with it.
comment vcObjectReleaseLater Puts the object in the release queue while its class's in_use callback says it is still in use; it keeps a reference there.
comment vcObjectReleaseQueued Goes over the release queue: lets go of the objects no longer in use, keeps the others.
comment vcObjectLoadChunk Reads one chunk of an object from a file (the file first, then the handle) through its class's load_chunk callback. Hands back 0 at the end of the chunks, 1 for a chunk read.
comment vcObjectSystemShutdown Ends the object system: the release queue, every class's table of instances and the classes go.
comment vcObjectSetName Names an object (lower case, at most 64 bytes). Refused if its class wants names unique and one of its objects has that name.
comment vcObjectGetShared The object of a class with that name, shared (one more reference) or cloned if the class wants copies. 0 if there is none.
comment vcObjectGet The object of a class with that name: the one loaded already, or read from its file, vcDataDir\xbox\CLASS\NAME.TAG.
comment vcObjectLoad Reads an object from a file: a reference to one read before in it, one the game has by name (shared, or read again), or a new one read through its class's load callback. Hands back its handle with a reference for the caller, 0 if none.
comment vcObjectReload Reads an object from a file into one that exists (the given handle, or the one of that name) through its class's reload callback.
comment vcDefault_ReturnZero What a class gives for a callback that has nothing to do: hands back 0.
comment vcDefault_ReturnOne What a class gives for a callback that only has to succeed: hands back 1.
comment vcDefault_ReturnArg3 What a class with no children gives for its collect callback: hands back the changed bits it was given.
